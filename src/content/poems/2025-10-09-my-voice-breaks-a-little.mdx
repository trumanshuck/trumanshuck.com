import { Box, Text } from 'rebass'

<Box sx={{whiteSpace: 'nowrap'}}>
  <Text>my voice breaks a little</Text>
  <Text>on the highest verse of shri ram</Text>
  <Text>as your arm stirs the air, brushing my</Text>
  <Text mb={3}>beard with one tiny hand in the dark</Text>

  <Text>you are the surest weight I've ever held</Text>
  <Text>and we are rocking slowly while I sing,</Text>
  <Text>every shifting moment presses your body into mine</Text>
  <Text mb={3}>as you mutter and yawn</Text>

  <Text>it's as if there could be nothing else, anywhere</Text>
  <Text>forgetting that there are dishes in the sink</Text>
  <Text mb={3}>and a world outside struggling to be reborn</Text>

  <Text>it was only half a year ago that you lay in your crib</Text>
  <Text>grasping my forearm with your whole body</Text>
  <Text>while I wiggled you to sleep</Text>
  <Text mb={3}>as if there could be anything else, ever</Text>

  <Text>but your cooing then has faded from my ears now,</Text>
  <Text mb={3}>and the marks of your small form will leave my lap someday empty.</Text>

  <Text>I slip away with a forehead kiss between verses</Text>
  <Text>but I am not left so vacant</Text>
  <Text>the gentle sway in this chair is not our last</Text>
  <Text>and who is to say which moment</Text>
  <Text mb={3}>impossibly, could be everything, always</Text>

  <Text>it must be this one, now</Text>
</Box>
